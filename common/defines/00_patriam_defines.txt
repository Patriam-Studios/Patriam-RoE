# Vanilla ends the game on 1453.1.1. This campaign opens in 6355 ADA, so
# without this the game declared itself over on the first day.
# Seven hundred years comfortably outlasts Hythaerion, who lives to about
# 175, and the reigns that follow him.
NGame = {
	END_DATE = "7055.1.1"
}

############################################################
# The long years of Patriam.
#
# Patriamic humans were made to hold their prime. A life runs at the ordinary
# rate until twenty, at a quarter of it from twenty to sixty, and at half of it
# afterwards. A man of forty is worn as a man of twenty five, one of sixty as
# thirty, one of a hundred as fifty, one of a hundred and twenty as sixty, and
# Hythaerion at 175 as a man of eighty seven.
#
# Every age that governs decline was mapped through that curve, and the health
# ramp was refitted to it rather than merely stretched, so the wear a character
# carries at any age matches the vanilla curve at the age above to within one
# per cent.
############################################################
